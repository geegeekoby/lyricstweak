# LyricsTweak -- build and publish.
#
# Paste this whole file into your GitHub repository as:
#
#     .github/workflows/build.yml
#
# It is the only file the repository needs. The tweak's source does not live on GitHub -- it lives
# in the git remote the library site deploys from, which a build runner cannot clone -- so this
# workflow asks the site for the source, builds the .deb on a macOS runner, and posts it back to
# the site's package repository. Sileo and Zebra then see the new build straight away: no tag, no
# commit, no redeploy.
#
# One secret is required, under Settings -> Secrets and variables -> Actions -> Secrets:
#
#     REPO_PUBLISH_TOKEN   the same value the site has for REPO_PUBLISH_TOKEN
#
# It authenticates both halves: fetching the source, and publishing the result. A repository
# variable of that name is not used and does not stand in for the secret.

name: build

on:
  push:
    branches: [main]
  workflow_dispatch:

permissions:
  contents: read

concurrency:
  group: build-${{ github.ref }}
  cancel-in-progress: true

jobs:
  tweak:
    name: tweak (.deb)
    runs-on: macos-14
    steps:
      - uses: actions/setup-node@v4
        with:
          node-version: "20"

      # The source, from the site this build publishes to. Unpacked with node rather than tar so
      # that nothing has to be installed before the source exists, and each path is checked to be
      # inside the workspace before it is written.
      - name: Fetch the tweak source
        env:
          REPO_PUBLISH_TOKEN: ${{ secrets.REPO_PUBLISH_TOKEN }}
          REPO_PUBLISH_URL: ${{ vars.REPO_PUBLISH_URL }}
        run: |
          set -euo pipefail
          if [ -z "${REPO_PUBLISH_TOKEN:-}" ]; then
            echo "REPO_PUBLISH_TOKEN is not set on this repository." >&2
            echo "Add it under Settings -> Secrets and variables -> Actions -> Secrets." >&2
            exit 1
          fi
          site="${REPO_PUBLISH_URL:-https://scintillating-pie-1f0e65.netlify.app}"
          site="${site%/}"
          curl -fsS --retry 3 --retry-delay 5 \
            -H "authorization: Bearer $REPO_PUBLISH_TOKEN" \
            "$site/repo/source" -o "$RUNNER_TEMP/source.json"
          node -e '
            const fs = require("fs");
            const path = require("path");
            const payload = JSON.parse(fs.readFileSync(process.argv[1], "utf8"));
            if (!payload.ok) throw new Error(payload.error || "the site refused the source request");
            const root = process.cwd();
            for (const [rel, b64] of Object.entries(payload.files)) {
              const out = path.resolve(root, rel);
              if (!out.startsWith(root + path.sep)) throw new Error("refusing path " + rel);
              fs.mkdirSync(path.dirname(out), { recursive: true });
              fs.writeFileSync(out, Buffer.from(b64, "base64"));
            }
            console.log("unpacked " + payload.count + " files, version " + payload.version);
          ' "$RUNNER_TEMP/source.json"
          rm -f "$RUNNER_TEMP/source.json"
          for required in Makefile control Tweak.xm scripts/publish-deb.mjs; do
            test -f "$required" || { echo "the site did not send $required" >&2; exit 1; }
          done

      # ldid signs the dylib, dpkg builds the archive, and xz is what dpkg-deb compresses with.
      # None of them ship with the runner.
      - name: Install packaging tools
        run: brew install ldid dpkg xz

      # /var/theos is where Theos installs by default and where the Makefile expects it.
      - name: Install Theos
        run: |
          set -euo pipefail
          sudo mkdir -p /var/theos
          sudo chown -R "$(whoami)" /var/theos
          git clone --quiet --recursive --depth 1 https://github.com/theos/theos.git /var/theos

      # Theos needs a patched SDK with the private headers Apple's own does not ship. These are
      # only compiled against; nothing private is linked.
      - name: Install iOS SDK
        run: |
          set -euo pipefail
          mkdir -p /var/theos/sdks
          git clone --quiet --depth 1 https://github.com/theos/sdks.git "$RUNNER_TEMP/theos-sdks"
          cp -r "$RUNNER_TEMP/theos-sdks/iPhoneOS16.5.sdk" /var/theos/sdks/
          rm -rf "$RUNNER_TEMP/theos-sdks"

      - name: Build
        run: |
          set -euo pipefail
          # FINALPACKAGE strips the binary and turns off debug logging; rootless is the packaging
          # scheme the Makefile already declares, restated so a change there is visible here.
          make package FINALPACKAGE=1 THEOS_PACKAGE_SCHEME=rootless
          ls -la packages

      # Kept so a build can be installed by hand as well, though it needs a logged-in GitHub
      # session to download -- which is exactly why publishing to the site is the step that
      # matters.
      - uses: actions/upload-artifact@v4
        with:
          name: LyricsTweak-deb
          path: packages/*.deb
          if-no-files-found: error

      # The site stores the bytes it is handed and measures those same bytes, so this is the whole
      # of publishing: Size and SHA256 in the package index describe exactly what the download
      # will hand back.
      - name: Publish to the repository
        env:
          REPO_PUBLISH_TOKEN: ${{ secrets.REPO_PUBLISH_TOKEN }}
          REPO_PUBLISH_URL: ${{ vars.REPO_PUBLISH_URL }}
        run: |
          set -euo pipefail
          deb="$(ls -t packages/*.deb | head -n1)"
          node scripts/publish-deb.mjs "$deb"
